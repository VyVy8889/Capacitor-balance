function clearInput() { document.getElementById('capacitor-input').value = ''; document.getElementById('results-section').classList.add('hidden'); currentSolution = null; parseAndValidate(false); } /** * Bộ đọc dữ liệu thông minh (Smart Parser): * - Hỗ trợ: "Tụ 38: 10.568", "Bình 1 - 10,54", "C01 = 4.12 uF" * - Hỗ trợ copy 2 cột từ Excel: "1 10.568" hoặc "38. 10.568" * - Hỗ trợ danh sách chỉ có số ngăn cách bằng xuống dòng, dấu phẩy, chấm phẩy hoặc tab */ function parseAndValidate(autoRun = true) { const inputEl = document.getElementById('capacitor-input'); const rawText = inputEl ? inputEl.value.trim() : ''; const cfg = getConfig(); parsedCapacitors = []; if (rawText.length > 0) { const lines = rawText.split(/\r?\n/); let autoIdx = 1; for (let line of lines) { const trimmed = line.trim(); if (!trimmed) continue; // 1. Dạng có dấu phân cách nhãn: ":", "=", " - ", " – ", "—" const kvMatch = trimmed.match(/^(.+?)\s*(?::|=|\s+[-–—]\s+)\s*([+-]?\d+(?:[.,]\d+)?)\s*(?:uF|µF|μF)?\s*$/i); if (kvMatch) { const label = kvMatch[1].trim(); const val = parseFloat(kvMatch[2].replace(',', '.')); if (!isNaN(val) && val > 0) { parsedCapacitors.push({ id: autoIdx++, label, value: val }); } continue; } // 2. Dạng 2 cột copy từ Excel (STT/Mã + Giá trị điện dung) ngăn cách bởi Tab hoặc nhiều khoảng trắng const colMatch = trimmed.match(/^([^\t\s]+(?:\s+[^\t\s]+)?)\s*(?:\t|\s{2,})\s*([+-]?\d+(?:[.,]\d+)?)\s*(?:uF|µF|μF)?\s*$/i); if (colMatch) { const col1 = colMatch[1].trim(); const val = parseFloat(colMatch[2].replace(',', '.')); if (!isNaN(val) && val > 0) { const label = /^\d+\.?$/.test(col1) ? `Tụ ${col1.replace('.', '')}` : col1; parsedCapacitors.push({ id: autoIdx++, label, value: val }); } continue; } // 3. Dạng số thứ tự đứng đầu: "1. 10.568" hoặc "38) 10.568" const numberedMatch = trimmed.match(/^(\d+)[.)]\s+([+-]?\d+(?:[.,]\d+)?)\s*(?:uF|µF|μF)?\s*$/i); if (numberedMatch) { const val = parseFloat(numberedMatch[2].replace(',', '.')); if (!isNaN(val) && val > 0) { parsedCapacitors.push({ id: autoIdx++, label: `Tụ ${numberedMatch[1]}`, value: val }); } continue; } // 4. Dạng chỉ chứa giá trị số (1 số trên dòng hoặc nhiều số cách nhau bởi tab, chấm phẩy, khoảng trắng) let tokens = []; if (trimmed.includes(';') || trimmed.includes('\t') || /\s+/.test(trimmed)) { tokens = trimmed.split(/[;\t\s]+/); } else if (trimmed.includes(',')) { const commaCount = (trimmed.match(/,/g) || []).length; if (commaCount > 1 || trimmed.includes('.')) { tokens = trimmed.split(','); } else { tokens = [trimmed]; } } else { tokens = [trimmed]; } for (let token of tokens) { const cleanNumStr = token.replace(/uF|µF|μF/gi, '').trim().replace(',', '.'); if (!cleanNumStr) continue; if (/^[+-]?\d+(?:\.\d+)?$/.test(cleanNumStr)) { const val = parseFloat(cleanNumStr); if (!isNaN(val) && val > 0) { const code = autoIdx < 10 ? `Tụ 0${autoIdx}` : `Tụ ${autoIdx}`; parsedCapacitors.push({ id: autoIdx++, label: code, value: val }); } } } } } // Cập nhật giao diện đếm và trạng thái nút Tính toán const count = parsedCapacitors.length; const detectedEl = document.getElementById('detected-total-count'); const boxEl = document.getElementById('validation-box'); const iconEl = document.getElementById('validation-icon'); const msgEl = document.getElementById('validation-message'); const btnEl = document.getElementById('calculate-btn'); detectedEl.textContent = count; if (count === 0) { detectedEl.className = 'text-blue-600'; boxEl.className = 'rounded-xl p-3 border text-xs flex items-start gap-2.5 bg-slate-50 border-slate-200 text-slate-600'; iconEl.className = 'fa-solid fa-circle-info text-slate-400 mt-0.5 text-sm'; msgEl.innerHTML = `Đang chờ nhập danh sách giá trị điện dung đo được (cần đủ ${cfg.totalReq} bình)...`; btnEl.disabled = true; } else if (count === cfg.totalReq) { detectedEl.className = 'text-emerald-600'; boxEl.className = 'rounded-xl p-3 border text-xs flex items-start gap-2.5 bg-emerald-50 border-emerald-200 text-emerald-800'; iconEl.className = 'fa-solid fa-circle-check text-emerald-600 mt-0.5 text-sm'; msgEl.innerHTML = `Đã nhận diện đủ ${count}/${cfg.totalReq} bình tụ! Sẵn sàng phân bổ ${cfg.nl} bình/nhánh Trái và ${cfg.nr} bình/nhánh Phải.`; btnEl.disabled = false; // Tự động chạy tính toán ngay khi vừa nhập/dán đủ số lượng yêu cầu if (autoRun) { runOptimization(true); } } else if (count < cfg.totalReq) { const missing = cfg.totalReq - count; detectedEl.className = 'text-amber-600'; boxEl.className = 'rounded-xl p-3 border text-xs flex items-start gap-2.5 bg-amber-50 border-amber-300 text-amber-900'; iconEl.className = 'fa-solid fa-triangle-exclamation text-amber-600 mt-0.5 text-sm'; msgEl.innerHTML = `Đã nhận diện ${count} bình. Còn thiếu ${missing} bình nữa để đủ ${cfg.totalReq} bình (${cfg.nl} Trái - ${cfg.nr} Phải).`; btnEl.disabled = true; } else { const extra = count - cfg.totalReq; detectedEl.className = 'text-rose-600'; boxEl.className = 'rounded-xl p-3 border text-xs flex items-start gap-2.5 bg-rose-50 border-rose-300 text-rose-900'; iconEl.className = 'fa-solid fa-circle-xmark text-rose-600 mt-0.5 text-sm'; msgEl.innerHTML = `Đã nhận diện ${count} bình (đang dư ${extra} bình so với yêu cầu ${cfg.totalReq} bình). Vui lòng xóa bớt ${extra} bình hoặc điều chỉnh số bình mỗi nhánh.`; btnEl.disabled = true; } } /** * Kích hoạt tính toán cân bằng */ function runOptimization(isAuto = false) { const cfg = getConfig(); if (parsedCapacitors.length !== cfg.totalReq) return; currentSolution = optimizeLeftRight(parsedCapacitors, cfg.nl, cfg.nr, 70); renderResults(); const resSec = document.getElementById('results-section'); resSec.classList.remove('hidden'); if (!isAuto) { resSec.scrollIntoView({ behavior: 'smooth', block: 'start' }); } showToast(`Đã cân bằng tối ưu 2 Vế & 3 Pha (${cfg.nl} Trái - ${cfg.nr} Phải)!`, 'success'); } // Lắng nghe đầy đủ sự kiện nhập/dán trên điện thoại (iOS/Android) và máy tính function bindInputEvents() { const inputEl = document.getElementById('capacitor-input'); if (!inputEl) return; ['input', 'change', 'keyup', 'blur'].forEach(evt => { inputEl.addEventListener(evt, () => parseAndValidate(true)); }); inputEl.addEventListener('paste', () => { setTimeout(() => parseAndValidate(true), 50); }); } // Khởi tạo mặc định cấu hình 2 Trái - 3 Phải (15 bình) và chạy thử mẫu khi tải trang window.onload = function() { bindInputEvents(); applyPreset(2, 3); generateMatchingSample(4.10); };
